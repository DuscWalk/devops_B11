            while i < len(lines) and (lines[i].startswith("	") or lines[i].strip() == ""):
                if lines[i].startswith("	"):
                    recipe.append(lines[i])
                i += 1
            rules.append(MakefileRule(
                target=target_part,
                prerequisites=prerequisites,
                recipe=recipe,
                line_no=rule_line,
            ))
        else:
            i += 1
    return rules


def detect_declaration_style(content, rules):
    style = {
        "uses_tab_recipe": any(r.recipe for r in rules),
        "prereq_separator": " ",
        "has_continuation": "\" in content,
        "variable_patterns": re.findall(r"$(([A-Za-z_]w*))", content),
    }
    return style


def find_rule_for_target(rules, target):
    for rule in rules:
        if target in rule.target.split():
            return rule
    return None


def generate_patch_for_missing(makefile_content, rules, missing):
    rule = find_rule_for_target(rules, missing.target)
    if rule is None:
        return None
    dep = missing.dependency
    if dep in rule.prerequisites:
        return None
    lines = makefile_content.split("
")
    rule_idx = rule.line_no - 1
    line = lines[rule_idx]
    colon_idx = line.index(":")
    target_part = line[:colon_idx]
    prereq_part = line[colon_idx + 1:]
    new_prereq = prereq_part.rstrip()
    if new_prereq and not new_prereq.endswith("\"):
        new_prereq = new_prereq + " " + dep
    elif new_prereq.endswith("\"):
        new_prereq = new_prereq + "
	" + dep + " \"
    else:
        new_prereq = " " + dep
    lines[rule_idx] = target_part + ":" + new_prereq
    patched = "
".join(lines)
    desc = f"add missing dependency {dep} to {missing.target}"
    return patched, desc
