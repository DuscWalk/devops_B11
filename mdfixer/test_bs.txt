def parse_artifact_uri(uri):
    m = re.match(r"^artifact://([w.-]+)/([w.-]+)/([w./-]+)$", uri)
    if not m:
        return None
    return m.group(1), m.group(2), m.group(3)


def resolve_artifact_path(uri):
    parsed = parse_artifact_uri(uri)
    if parsed is None:
        return None
    group, job_id, name = parsed
    if not ARTIFACT_ROOT:
        return None
    return Path(ARTIFACT_ROOT) / group / job_id / name


def parse_makefile_uri(uri):
    m = re.match(r"^repo://([0-9a-f]{40})/(.+)$", uri)
    if not m:
        return None
    return m.group(1), m.group(2)


def load_md_report(md_report_uri):
    path = resolve_artifact_path(md_report_uri)
    if path is None or not path.exists():
        p = Path(md_report_uri)
        if p.exists():
            path = p
        else:
            raise FileNotFoundError(f"cannot resolve md_report_uri: {md_report_uri}")
    with open(path, "r", encoding="utf-8") as f:
        data = json.load(f)
    findings_raw = data
    if isinstance(data, dict):
        if "output" in data and isinstance(data["output"], dict):
            findings_raw = data["output"].get("findings", [])
        elif "findings" in data:
            findings_raw = data["findings"]
        else:
            findings_raw = []
    findings = []
    for item in findings_raw:
        findings.append(Finding(
            type=item.get("type", ""),
            target=item.get("target", ""),
            dependency=item.get("dependency", ""),
            commit=item.get("commit", ""),
            detector=item.get("detector", ""),
            location=item.get("location", {}),
            evidence=item.get("evidence", {}),
            suggested_fix=item.get("suggested_fix", ""),
        ))
    return findings


def filter_missing(findings):
    return [f for f in findings if f.type == "MISSING"]


@dataclass
class MakefileRule:
    target: str
    prerequisites: list[str]
    recipe: list[str]
    line_no: int


def parse_makefile(content):
    rules = []
    lines = content.split("
")
    i = 0
    while i < len(lines):
        line = lines[i]
        stripped = line.strip()
        if not stripped or stripped.startswith("#"):
            i += 1
            continue
        if re.match(r"^[A-Za-z_]w*s*[:?+]?=", line) and ":" not in line.split("=", 1)[0]:
            i += 1
            continue
        if not line.startswith("	") and ":" in line:
            colon_idx = line.index(":")
            if colon_idx + 1 < len(line) and line[colon_idx + 1] == "=":
                i += 1
                continue
            target_part = line[:colon_idx].strip()
            prereq_part = line[colon_idx + 1:].strip()
            while prereq_part.endswith("\"):
                prereq_part = prereq_part[:-1].rstrip()
                i += 1
                if i < len(lines):
                    prereq_part += " " + lines[i].strip()
            prerequisites = prereq_part.split() if prereq_part else []
            recipe = []
            rule_line = i + 1
            i += 1
