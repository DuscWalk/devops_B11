def parse_makefile(content):
    rules = []
    lines = content.split("
")
    i = 0
    while i < len(lines):
        line = lines[i]
        stripped = line.strip()
        if not stripped or stripped.startswith("#"):
            i += 1
            continue
        if re.match(r"^[A-Za-z_]w*s*[:?+]?=", line) and ":" not in line.split("=", 1)[0]:
            i += 1
            continue
        if not line.startswith("	") and ":" in line:
            colon_idx = line.index(":")
            if colon_idx + 1 < len(line) and line[colon_idx + 1] == "=":
                i += 1
                continue
            target_part = line[:colon_idx].strip()
            prereq_part = line[colon_idx + 1:].strip()
            while prereq_part.endswith("\"):
                prereq_part = prereq_part[:-1].rstrip()
                i += 1
                if i < len(lines):
                    prereq_part += " " + lines[i].strip()
            prerequisites = prereq_part.split() if prereq_part else []
            recipe = []
            rule_line = i + 1
            i += 1
            while i < len(lines) and (lines[i].startswith("	") or lines[i].strip() == ""):
                if lines[i].startswith("	"):
                    recipe.append(lines[i])
                i += 1
            rules.append(MakefileRule(
                target=target_part,
                prerequisites=prerequisites,
                recipe=recipe,
                line_no=rule_line,
            ))
        else:
            i += 1
    return rules


def detect_declaration_style(content, rules):
    style = {
        "uses_tab_recipe": any(r.recipe for r in rules),
        "prereq_separator": " ",
        "has_continuation": "\" in content,
        "variable_patterns": re.findall(r"$(([A-Za-z_]w*))", content),
    }
    return style


def find_rule_for_target(rules, target):
    for rule in rules:
        if target in rule.target.split():
            return rule
    return None


def generate_patch_for_missing(makefile_content, rules, missing):
    rule = find_rule_for_target(rules, missing.target)
    if rule is None:
        return None
    dep = missing.dependency
    if dep in rule.prerequisites:
        return None
    lines = makefile_content.split("
")
    rule_idx = rule.line_no - 1
    line = lines[rule_idx]
    colon_idx = line.index(":")
    target_part = line[:colon_idx]
    prereq_part = line[colon_idx + 1:]
    new_prereq = prereq_part.rstrip()
    if new_prereq and not new_prereq.endswith("\"):
        new_prereq = new_prereq + " " + dep
    elif new_prereq.endswith("\"):
        new_prereq = new_prereq + "
	" + dep + " \"
    else:
        new_prereq = " " + dep
    lines[rule_idx] = target_part + ":" + new_prereq
    patched = "
".join(lines)
    desc = f"add missing dependency {dep} to {missing.target}"
    return patched, desc


def make_unified_diff(original, patched, filename):
    orig_lines = original.splitlines(keepends=True)
    patched_lines = patched.splitlines(keepends=True)
    diff = difflib.unified_diff(
        orig_lines,
        patched_lines,
        fromfile=f"a/{filename}",
        tofile=f"b/{filename}",
        lineterm="",
    )
    return "".join(diff)
